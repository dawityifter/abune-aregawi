# Structural disagreements between the JPA entities and the schema the Node
# backend's Sequelize migrations own. Each line is a column an entity declares
# that no table has, so any query touching it fails at runtime.
#
# SchemaGapTest fails when this list GROWS. Shrinking it is the work; when a
# line is fixed, delete it from here in the same commit.
#
# Cosmetic type mismatches (varchar(255) vs varchar(100) and similar) are
# deliberately excluded — there are ~50 and Hibernate handles them fine.
#
# Regenerate: ./gradlew test --tests '*SchemaGapReportTest*' then read
# build/schema-gap.sql.
alter table if exists members add column allergies TEXT
alter table if exists members add column dietary_restrictions TEXT
alter table if exists members add column is_baptized boolean
alter table if exists members add column medical_conditions TEXT
alter table if exists members add column medications TEXT
alter table if exists members add column notes TEXT
alter table if exists pledges add column status varchar(255) not null check ((status in ('pending','fulfilled','expired','cancelled')))
